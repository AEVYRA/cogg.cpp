cmake_minimum_required(VERSION 3.20)
project(cogg VERSION 0.2.0 LANGUAGES CXX)
option(COGG_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(COGG_LLAMA "Build the pinned local llama.cpp backend" OFF)
set(COGG_TEST_MODEL "" CACHE FILEPATH "Optional GGUF for real inference integration tests")
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Threads REQUIRED)
add_library(cogg src/kernel.cpp)
target_include_directories(cogg PUBLIC include)
target_compile_features(cogg PUBLIC cxx_std_20)
target_link_libraries(cogg PUBLIC nlohmann_json::nlohmann_json PRIVATE SQLite::SQLite3 OpenSSL::Crypto)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cogg PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(COGG_SANITIZE)
    target_compile_options(cogg PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cogg PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(cogg-cli tools/main.cpp)
target_link_libraries(cogg-cli PRIVATE cogg)
if(COGG_LLAMA)
  include(FetchContent)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(llama
    URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/0cae43063cf15170e91a2ff4d034da0ecef4a1b2
    URL_HASH SHA256=5425569b37eb461bf82feba00f765fdb8624d7a0bf4719320282d6f0895f379f)
  FetchContent_MakeAvailable(llama)
  add_library(cogg-llama src/llama_backend.cpp)
  target_link_libraries(cogg-llama PUBLIC cogg PRIVATE llama OpenSSL::Crypto)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cogg-llama PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  target_link_libraries(cogg-cli PRIVATE cogg-llama)
  target_compile_definitions(cogg-cli PRIVATE COGG_LLAMA)
  install(TARGETS cogg-llama)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(cogg-tests tests/kernel_test.cpp)
  target_link_libraries(cogg-tests PRIVATE cogg SQLite::SQLite3 Threads::Threads)
  add_test(NAME kernel COMMAND cogg-tests)
  if(COGG_LLAMA)
    add_executable(cogg-llama-tests tests/llama_test.cpp)
    target_link_libraries(cogg-llama-tests PRIVATE cogg-llama)
    add_test(NAME llama-options COMMAND cogg-llama-tests)
    if(COGG_TEST_MODEL)
      add_test(NAME llama-inference COMMAND cogg-llama-tests "${COGG_TEST_MODEL}")
      set_tests_properties(llama-inference PROPERTIES TIMEOUT 600)
    endif()
  endif()
  if(UNIX)
    add_executable(cogg-crash-tests tests/crash_test.cpp)
    target_link_libraries(cogg-crash-tests PRIVATE cogg)
    add_test(NAME process-crash COMMAND cogg-crash-tests)
    set_tests_properties(process-crash PROPERTIES TIMEOUT 30)
  endif()
endif()
install(TARGETS cogg cogg-cli)
install(DIRECTORY include/cogg DESTINATION include)
