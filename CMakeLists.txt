cmake_minimum_required(VERSION 3.20)
project(cogg VERSION 0.8.1 LANGUAGES CXX)
option(COGG_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(COGG_HTTP "Build Ollama and Chat Completions adapters" OFF)
option(COGG_SELF "Build the optional self-state host policy" OFF)
option(COGG_LLAMA "Build the pinned local llama.cpp backend" OFF)
set(COGG_TEST_MODEL "" CACHE FILEPATH "Optional GGUF for real inference integration tests")
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Threads REQUIRED)
add_library(cogg src/kernel.cpp src/checkpoint.cpp src/memory.cpp src/routing.cpp src/outcome.cpp)
target_include_directories(cogg PUBLIC include)
target_compile_features(cogg PUBLIC cxx_std_20)
target_link_libraries(cogg PUBLIC nlohmann_json::nlohmann_json PRIVATE SQLite::SQLite3 OpenSSL::Crypto)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cogg PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(COGG_SANITIZE)
    target_compile_options(cogg PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cogg PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(cogg-cli tools/main.cpp)
target_link_libraries(cogg-cli PRIVATE cogg)
if(COGG_LLAMA)
  include(FetchContent)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(llama
    URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/0cae43063cf15170e91a2ff4d034da0ecef4a1b2
    URL_HASH SHA256=5425569b37eb461bf82feba00f765fdb8624d7a0bf4719320282d6f0895f379f)
  FetchContent_MakeAvailable(llama)
  add_library(cogg-llama src/llama_backend.cpp)
  # A changed adapter/compiler/build mode must not silently reinterpret opaque KV.
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/llama_backend.cpp)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/llama_backend.cpp" COGG_ADAPTER_HASH)
  string(SHA256 COGG_CHECKPOINT_BUILD "${COGG_ADAPTER_HASH};${CMAKE_SYSTEM_NAME};${CMAKE_SYSTEM_PROCESSOR};${CMAKE_CXX_COMPILER_ID};${CMAKE_CXX_COMPILER_VERSION};${CMAKE_C_COMPILER_ID};${CMAKE_C_COMPILER_VERSION};${CMAKE_CXX_FLAGS};${CMAKE_CXX_FLAGS_RELEASE};${CMAKE_CXX_FLAGS_DEBUG};${CMAKE_CXX_FLAGS_RELWITHDEBINFO};${CMAKE_C_FLAGS};${CMAKE_C_FLAGS_RELEASE};${CMAKE_C_FLAGS_DEBUG};${CMAKE_C_FLAGS_RELWITHDEBINFO};${CMAKE_BUILD_TYPE};${COGG_SANITIZE}")
  target_compile_definitions(cogg-llama PRIVATE COGG_CHECKPOINT_BUILD="${COGG_CHECKPOINT_BUILD}")
  target_link_libraries(cogg-llama PUBLIC cogg PRIVATE llama OpenSSL::Crypto)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cogg-llama PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  target_link_libraries(cogg-cli PRIVATE cogg-llama)
  target_compile_definitions(cogg-cli PRIVATE COGG_LLAMA)
  install(TARGETS cogg-llama)
endif()
if(COGG_HTTP)
  find_package(CURL 7.85 REQUIRED)
  add_library(cogg-http src/http_backend.cpp)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/http_backend.cpp)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/http_backend.cpp" COGG_HTTP_ADAPTER_HASH)
  target_compile_definitions(cogg-http PRIVATE COGG_HTTP_ADAPTER_HASH="${COGG_HTTP_ADAPTER_HASH}")
  target_link_libraries(cogg-http PUBLIC cogg PRIVATE CURL::libcurl)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cogg-http PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  target_link_libraries(cogg-cli PRIVATE cogg-http)
  target_compile_definitions(cogg-cli PRIVATE COGG_HTTP)
  install(TARGETS cogg-http)
endif()
if(COGG_SELF)
  add_library(cogg-self src/self_state.cpp)
  target_link_libraries(cogg-self PUBLIC cogg)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cogg-self PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  if(COGG_HTTP)
    add_executable(cogg-self-example tools/self_example.cpp)
    target_link_libraries(cogg-self-example PRIVATE cogg-self cogg-http)
  endif()
  install(TARGETS cogg-self)
endif()
include(CTest)
if(BUILD_TESTING)
  if(COGG_SELF)
    add_executable(cogg-self-tests tests/self_state_test.cpp)
    target_link_libraries(cogg-self-tests PRIVATE cogg-self)
    add_test(NAME self-state COMMAND cogg-self-tests)
    if(COGG_HTTP)
      find_package(Python3 REQUIRED COMPONENTS Interpreter)
      add_test(NAME self-process COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/self_process_test.py $<TARGET_FILE:cogg-self-example> $<TARGET_FILE:cogg-cli>)
      set_tests_properties(self-process PROPERTIES TIMEOUT 60)
    endif()
  endif()
  add_executable(cogg-outcome-tests tests/outcome_test.cpp)
  target_link_libraries(cogg-outcome-tests PRIVATE cogg SQLite::SQLite3)
  add_test(NAME outcomes COMMAND cogg-outcome-tests)
  add_executable(cogg-routing-tests tests/routing_test.cpp)
  target_link_libraries(cogg-routing-tests PRIVATE cogg)
  add_test(NAME routing COMMAND cogg-routing-tests)
  if(COGG_HTTP)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME http-faults COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py $<TARGET_FILE:cogg-cli>)
    set_tests_properties(http-faults PROPERTIES TIMEOUT 90)
  endif()
  add_executable(cogg-memory-benchmark tools/memory_benchmark.cpp)
  target_link_libraries(cogg-memory-benchmark PRIVATE cogg)
  add_executable(cogg-memory-tests tests/memory_test.cpp)
  target_link_libraries(cogg-memory-tests PRIVATE cogg SQLite::SQLite3)
  target_compile_definitions(cogg-memory-tests PRIVATE COGG_LEGACY_V2_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/legacy-v2.sql")
  add_test(NAME memory COMMAND cogg-memory-tests)
  add_executable(cogg-tests tests/kernel_test.cpp)
  target_link_libraries(cogg-tests PRIVATE cogg SQLite::SQLite3 Threads::Threads)
  add_test(NAME kernel COMMAND cogg-tests)
  add_executable(cogg-time-tests tests/time_test.cpp)
  target_link_libraries(cogg-time-tests PRIVATE cogg SQLite::SQLite3)
  target_compile_definitions(cogg-time-tests PRIVATE COGG_LEGACY_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/legacy-v1.sql")
  add_test(NAME temporal COMMAND cogg-time-tests)
  if(COGG_LLAMA)
    add_executable(cogg-llama-tests tests/llama_test.cpp)
    target_link_libraries(cogg-llama-tests PRIVATE cogg-llama)
    add_test(NAME llama-options COMMAND cogg-llama-tests)
    if(COGG_TEST_MODEL)
      add_executable(cogg-llama-outcome-tests tests/llama_outcome_test.cpp)
      target_link_libraries(cogg-llama-outcome-tests PRIVATE cogg-llama)
      add_test(NAME llama-outcome COMMAND cogg-llama-outcome-tests "${COGG_TEST_MODEL}")
      set_tests_properties(llama-outcome PROPERTIES TIMEOUT 180)
      add_executable(cogg-memory-model-tests tests/memory_model_test.cpp)
      target_link_libraries(cogg-memory-model-tests PRIVATE cogg-llama)
      add_test(NAME memory-model COMMAND cogg-memory-model-tests "${COGG_TEST_MODEL}")
      set_tests_properties(memory-model PROPERTIES TIMEOUT 600)
      add_executable(cogg-llama-checkpoint-tests tests/llama_checkpoint_test.cpp)
      target_link_libraries(cogg-llama-checkpoint-tests PRIVATE cogg-llama)
      add_test(NAME llama-inference COMMAND cogg-llama-tests "${COGG_TEST_MODEL}")
      add_test(NAME llama-checkpoint COMMAND cogg-llama-checkpoint-tests "${COGG_TEST_MODEL}")
      set_tests_properties(llama-inference PROPERTIES TIMEOUT 600)
      set_tests_properties(llama-checkpoint PROPERTIES TIMEOUT 600)
    endif()
  endif()
  if(UNIX)
    add_executable(cogg-checkpoint-tests tests/checkpoint_test.cpp)
    target_link_libraries(cogg-checkpoint-tests PRIVATE cogg)
    add_test(NAME checkpoint COMMAND cogg-checkpoint-tests)
    set_tests_properties(checkpoint PROPERTIES TIMEOUT 30)
    add_executable(cogg-crash-tests tests/crash_test.cpp)
    target_link_libraries(cogg-crash-tests PRIVATE cogg)
    add_test(NAME process-crash COMMAND cogg-crash-tests)
    set_tests_properties(process-crash PROPERTIES TIMEOUT 30)
  endif()
endif()
install(TARGETS cogg cogg-cli)
install(DIRECTORY include/cogg DESTINATION include)
