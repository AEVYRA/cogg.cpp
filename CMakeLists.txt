cmake_minimum_required(VERSION 3.20)
project(cogg VERSION 0.1.0 LANGUAGES CXX)
option(COGG_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Threads REQUIRED)
add_library(cogg src/kernel.cpp)
target_include_directories(cogg PUBLIC include)
target_compile_features(cogg PUBLIC cxx_std_20)
target_link_libraries(cogg PUBLIC nlohmann_json::nlohmann_json PRIVATE SQLite::SQLite3 OpenSSL::Crypto)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cogg PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(COGG_SANITIZE)
    target_compile_options(cogg PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cogg PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(cogg-cli tools/main.cpp)
target_link_libraries(cogg-cli PRIVATE cogg)
include(CTest)
if(BUILD_TESTING)
  add_executable(cogg-tests tests/kernel_test.cpp)
  target_link_libraries(cogg-tests PRIVATE cogg SQLite::SQLite3 Threads::Threads)
  add_test(NAME kernel COMMAND cogg-tests)
  if(UNIX)
    add_executable(cogg-crash-tests tests/crash_test.cpp)
    target_link_libraries(cogg-crash-tests PRIVATE cogg)
    add_test(NAME process-crash COMMAND cogg-crash-tests)
    set_tests_properties(process-crash PROPERTIES TIMEOUT 30)
  endif()
endif()
install(TARGETS cogg cogg-cli)
install(DIRECTORY include/cogg DESTINATION include)
